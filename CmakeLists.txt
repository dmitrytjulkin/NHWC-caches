cmake_minimum_required (VERSION 3.20)
project (nhwc_project LANGUAGES CXX)
set (CMAKE_CXX_STANDARD_REQUIRED ON)

set (COMMON_SRCS get_params.cpp slow_get_page.cpp)

add_library (cmmn_lib STATIC ${COMMON_SRCS})
target_include_directories (cmmn_lib PUBLIC . LIRS)
target_compile_features (cmmn_lib PRIVATE cxx_std_17)

add_executable (nhwc.x main.cpp)
target_link_libraries (nhwc.x cmmn_lib)
target_compile_features (nhwc.x PRIVATE cxx_std_17)

enable_testing ()

option (BUILD_TESTS "Build unit-tests for all algorithms" OFF)

function (add_testrun SUB_DIR)
    if (BUILD_TESTS)
        add_subdirectory (${SUB_DIR})
    endif()
endfunction ()

add_testrun (LFU)

add_testrun (LIRS)
